import java.util.*;

class ThroneInheritance {

    // Stores children of each person
    HashMap<String, List<String>> family;

    // Stores whether a person is dead
    HashSet<String> dead;

    String king;

    public ThroneInheritance(String kingName) {
        king = kingName;
        family = new HashMap<>();
        dead = new HashSet<>();

        family.put(kingName, new ArrayList<>());
    }

    public void birth(String parentName, String childName) {

        // Add child to parent's children list
        family.get(parentName).add(childName);

        // Create empty children list for the new child
        family.put(childName, new ArrayList<>());
    }

    public void death(String name) {

        // Mark person as dead
        dead.add(name);
    }

    public List<String> getInheritanceOrder() {

        List<String> order = new ArrayList<>();

        dfs(king, order);

        return order;
    }

    private void dfs(String person, List<String> order) {

        // Add person if alive
        if (!dead.contains(person)) {
            order.add(person);
        }

        // Visit children in birth order
        for (String child : family.get(person)) {
            dfs(child, order);
        }
    }
}
