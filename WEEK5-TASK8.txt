import java.util.*;

class Solution {
    public List<String> findAndReplacePattern(String[] words, String pattern) {
        List<String> result = new ArrayList<>();

        for (String word : words) {
            int[] a = new int[26];
            int[] b = new int[26];
            boolean match = true;

            for (int i = 0; i < pattern.length(); i++) {
                int x = pattern.charAt(i) - 'a';
                int y = word.charAt(i) - 'a';

                if (a[x] != b[y]) {
                    match = false;
                    break;
                }

                a[x] = i + 1;
                b[y] = i + 1;
            }

            if (match) {
                result.add(word);
            }
        }

        return result;
    }
}
